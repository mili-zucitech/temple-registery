1. PURPOSE

   Define the canonical documents table, all per-module junction tables,
   the document_access_log table, and the S3 object key structure.

--------------------------------------------------------------------------------

3. DETAILED CONTENT

   3.1 documents - CANONICAL METADATA TABLE

   CREATE TABLE documents (
     id                 BIGINT UNSIGNED  NOT NULL AUTO_INCREMENT,
     original_filename  VARCHAR(500)     NOT NULL,
     s3_key             VARCHAR(1000)    NOT NULL,
     s3_bucket          VARCHAR(100)     NOT NULL,
     mime_type          VARCHAR(100)     NOT NULL,        -- must be application/pdf
     file_size_bytes    INT UNSIGNED     NOT NULL,        -- max 10,485,760 (10 MB)
     module_reference   ENUM('TEMPLE','TRUST','EMPLOYEE',
                             'CONTRACTOR','DECLARATION')  NOT NULL,
     uploaded_by        BIGINT UNSIGNED  NOT NULL,
     uploaded_at        DATETIME(6)      NOT NULL DEFAULT CURRENT_TIMESTAMP(6),
     created_at         DATETIME(6)      NOT NULL DEFAULT CURRENT_TIMESTAMP(6),
     updated_at         DATETIME(6)      NOT NULL DEFAULT CURRENT_TIMESTAMP(6)
                                         ON UPDATE CURRENT_TIMESTAMP(6),
     created_by         BIGINT UNSIGNED  NOT NULL,
     updated_by         BIGINT UNSIGNED  NOT NULL,
     PRIMARY KEY (id),
     UNIQUE KEY uq_s3_key (s3_key),
     INDEX idx_doc_module (module_reference, uploaded_at),
     INDEX idx_doc_uploader (uploaded_by, uploaded_at)
   ) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

   3.2 PER-MODULE JUNCTION TABLES

   -- Temple documents
   CREATE TABLE temple_documents (
     id            BIGINT UNSIGNED NOT NULL AUTO_INCREMENT,
     temple_id     BIGINT UNSIGNED NOT NULL,
     document_id   BIGINT UNSIGNED NOT NULL,
     document_type VARCHAR(100)    NOT NULL,
     created_at    DATETIME(6) NOT NULL DEFAULT CURRENT_TIMESTAMP(6),
     updated_at    DATETIME(6) NOT NULL DEFAULT CURRENT_TIMESTAMP(6)
                               ON UPDATE CURRENT_TIMESTAMP(6),
     created_by    BIGINT UNSIGNED NOT NULL,
     updated_by    BIGINT UNSIGNED NOT NULL,
     PRIMARY KEY (id),
     UNIQUE KEY uq_temple_doc (temple_id, document_id),
     CONSTRAINT fk_td_temple   FOREIGN KEY (temple_id)   REFERENCES temples(id),
     CONSTRAINT fk_td_document FOREIGN KEY (document_id) REFERENCES documents(id)
   ) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

   -- Trust documents
   CREATE TABLE trust_documents (
     id            BIGINT UNSIGNED NOT NULL AUTO_INCREMENT,
     trust_id      BIGINT UNSIGNED NOT NULL,
     document_id   BIGINT UNSIGNED NOT NULL,
     document_type VARCHAR(100)    NOT NULL,
     created_at    DATETIME(6) NOT NULL DEFAULT CURRENT_TIMESTAMP(6),
     updated_at    DATETIME(6) NOT NULL DEFAULT CURRENT_TIMESTAMP(6)
                               ON UPDATE CURRENT_TIMESTAMP(6),
     created_by    BIGINT UNSIGNED NOT NULL,
     updated_by    BIGINT UNSIGNED NOT NULL,
     PRIMARY KEY (id),
     UNIQUE KEY uq_trust_doc (trust_id, document_id),
     CONSTRAINT fk_trd_trust    FOREIGN KEY (trust_id)    REFERENCES trust_registrations(id),
     CONSTRAINT fk_trd_document FOREIGN KEY (document_id) REFERENCES documents(id)
   ) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

   -- Employee documents
   CREATE TABLE employee_documents (
     id            BIGINT UNSIGNED NOT NULL AUTO_INCREMENT,
     employee_id   BIGINT UNSIGNED NOT NULL,
     document_id   BIGINT UNSIGNED NOT NULL,
     document_type VARCHAR(100)    NOT NULL,
     created_at    DATETIME(6) NOT NULL DEFAULT CURRENT_TIMESTAMP(6),
     updated_at    DATETIME(6) NOT NULL DEFAULT CURRENT_TIMESTAMP(6)
                               ON UPDATE CURRENT_TIMESTAMP(6),
     created_by    BIGINT UNSIGNED NOT NULL,
     updated_by    BIGINT UNSIGNED NOT NULL,
     PRIMARY KEY (id),
     UNIQUE KEY uq_emp_doc (employee_id, document_id),
     CONSTRAINT fk_ed_employee FOREIGN KEY (employee_id) REFERENCES employees(id),
     CONSTRAINT fk_ed_document FOREIGN KEY (document_id) REFERENCES documents(id)
   ) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

   -- Declaration documents
   CREATE TABLE declaration_documents (
     id             BIGINT UNSIGNED NOT NULL AUTO_INCREMENT,
     declaration_id BIGINT UNSIGNED NOT NULL,
     document_id    BIGINT UNSIGNED NOT NULL,

     created_at     DATETIME(6) NOT NULL DEFAULT CURRENT_TIMESTAMP(6),
     updated_at     DATETIME(6) NOT NULL DEFAULT CURRENT_TIMESTAMP(6)

     PRIMARY KEY (id),

       REFERENCES asset_declarations(id),
     CONSTRAINT fk_dd_document FOREIGN KEY (document_id) REFERENCES documents(id)
   ) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

   3.3 document_access_log - PRE-SIGNED URL AUDIT

   CREATE TABLE document_access_log (
     id                    BIGINT UNSIGNED NOT NULL AUTO_INCREMENT,
     document_id           BIGINT UNSIGNED NOT NULL,
     accessed_by           BIGINT UNSIGNED NOT NULL,
     accessed_at           DATETIME(6)     NOT NULL DEFAULT CURRENT_TIMESTAMP(6),
     ip_address            VARCHAR(45)     DEFAULT NULL,
     presigned_url_expiry  DATETIME(6)     NOT NULL,
     PRIMARY KEY (id),
     INDEX idx_dal_document (document_id, accessed_at),
     INDEX idx_dal_user (accessed_by, accessed_at),
     CONSTRAINT fk_dal_document FOREIGN KEY (document_id) REFERENCES documents(id)
   ) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

   RULE: INSERT only. Written BEFORE the pre-signed URL is returned to caller.

   3.4 S3 KEY STRUCTURE

   Pattern: {module}/{temple_id}/{year_month}/{uuid}.pdf

   Examples:
     temple/42/2026-04/a1b2c3d4-e5f6-7890-abcd-ef1234567890.pdf
     declaration/42/2026-04/b2c3d4e5-f6a7-8901-bcde-f12345678901.pdf
     employee/42/2026-04/c3d4e5f6-a7b8-9012-cdef-123456789012.pdf

   Rules:
   - UUID v4 as filename (never use original filename as S3 key)
   - All keys under one private bucket: trm-documents-prod (or trm-documents-staging)
   - Pre-signed URL TTL: 15 minutes maximum
   - Never generate public S3 URLs

   3.5 UPLOAD PIPELINE (SERVER-SIDE VALIDATION)

   Step 1: Receive multipart/form-data upload
   Step 2: Read first 4 bytes (magic bytes) - must be: 25 50 44 46 (%PDF)
   Step 3: Validate Content-Type header = application/pdf (secondary check)
   Step 4: Validate file size <= 10,485,760 bytes (10 MB)
   Step 5: Generate UUID key; upload to S3 private bucket
   Step 6: Insert row into documents table
   Step 7: Insert row into appropriate junction table
   Step 8: Return document ID (not S3 key) to frontend

   Download pipeline:
   Step 1: Verify caller has permission to access this document_id
             (join to junction table and verify temple_id / district_id scope)
   Step 2: Generate pre-signed URL (TTL = 900 seconds / 15 minutes)
   Step 3: Insert row into document_access_log
   Step 4: Return pre-signed URL to client

--------------------------------------------------------------------------------

4. DECISIONS LOCKED

   - MIME validation via magic bytes (server-side); reject on mismatch
   - S3 key = UUID (never original filename)
   - Pre-signed URL TTL = 15 minutes maximum
   - document_access_log written BEFORE URL is returned (not after)
   - documents table retains S3 key even if document is "deleted" (set a
     logical is_active flag; do not delete the row or the S3 object)

--------------------------------------------------------------------------------

5. OPEN QUESTIONS

   - Should S3 object versioning be enabled on the bucket?
     Pro: protects against accidental overwrites. Con: storage cost.
   - Should virus scanning (ClamAV or AWS Macie) run on uploaded PDFs?

--------------------------------------------------------------------------------

6. RISKS & EDGE CASES

   - Pre-signed URL can be forwarded to unauthorised party. The URL itself is
     the credential. Mitigate by short TTL (15 min) and access log monitoring.
   - S3 bucket ACL must be private + Block Public Access enabled at bucket and
     account level. Verify in IaC before go-live.

--------------------------------------------------------------------------------

7. ASSUMPTIONS

   - One S3 bucket per environment (prod, staging, dev) with consistent naming
   - All documents are PDFs; no image or video uploads in scope for v1
   - document_access_log is append-only; no archival needed for v1
